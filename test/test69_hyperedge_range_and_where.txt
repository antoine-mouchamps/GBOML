// A hyperedge's `for` range and its `where` read the edge's own parameters.
// Neither was visited by the naming pass, so an edge range or condition could
// name nothing but the time horizon.
#TIMEHORIZON
T = 6;

#NODE A
#VARIABLES
external : y[6];
#CONSTRAINTS
fix : y[i] == i + 1 for i in [0:5];
#OBJECTIVES
min : y[0];

#NODE B
#VARIABLES
external : w[6];
#CONSTRAINTS
lo : w[i] >= 0 for i in [0:5];
#OBJECTIVES
min cost : w[i] for i in [0:5];

#HYPEREDGE H
#PARAMETERS
first = 1;
jump  = 2;
last  = 5;
skip  = 3;
#CONSTRAINTS
// k runs over 1, 3 and 5, and the condition drops 3, so w[1] and w[5] are the
// only ones pinned to y : every other one falls to its lower bound
link : B.w[k] == A.y[k] for k in [first:jump:last] where k != skip;
