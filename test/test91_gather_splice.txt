            internal : prod[KSEP][T][N];
            internal : waste[T][N];

            waste[t][n] == 9;
    #VARIABLES

        internal : other[4][T][N] <- {P.waste[T][N], P.prod[KSEP][T][N]};
        internal : whole[KSEP][T][N] <- {P.prod[KSEP][T][N]};
    #OBJECTIVES
