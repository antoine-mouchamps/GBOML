#TIMEHORIZON
T = 2;

#NODE A
#PARAMETERS
c = {10, 20};
#VARIABLES
internal : x[2];
internal : z[2];
#CONSTRAINTS
lo_x: x[k] >= 1 for k in [0:1];
up_x: x[k] <= 3 for k in [0:1];
lo_z: z[k] >= 0 for k in [0:1];
up_z: z[k] <= 99 for k in [0:1];
// the index of the sum shadows the one of the constraint, the z[k] of the
// right hand side must still be expanded over the index of the constraint
squares: sum(x[k]*x[k] for k in [0:1]) == z[k] for k in [0:1];
#OBJECTIVES
min cost: c[k]*z[k] for k in [0:1];
