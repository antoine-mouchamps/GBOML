// Sum semantics : every shape of range the language accepts, with x[i] pinned
// to i+1 so each z is a single known value. Three of these were silently wrong
// before the factorizer rewrite, see KNOWN_ISSUES.md
#TIMEHORIZON
T = 4;

#NODE A
#PARAMETERS
N = 4;
#VARIABLES
internal : x[N];
internal : z_ragged[T];
internal : z_cond[T];
internal : z_ragged_cond[T];
internal : z_nested[T];
internal : z_nested_ragged[T];
internal : z_body_t[T];
#CONSTRAINTS
fix_x       : x[i] == i + 1 for i in [0:N-1];
// the range ends on the index of the constraint, so every row is shorter
// than the next one : [1, 3, 6, 10]
ragged      : z_ragged[t] == sum(x[i] for i in [0:t]);
// a where condition drops i == 1 : 1 + 3 + 4 == 8 on every row
cond        : z_cond[t] == sum(x[i] for i in [0:N-1] where i != 1);
// both at once, which is the case that has to keep the rows rectangular
// once the condition dropped entries : [1, 1, 4, 8]
ragged_cond : z_ragged_cond[t] == sum(x[i] for i in [0:t] where i != 1);
// a sum inside a sum, the inner range depending on the outer index :
// 1 + 3 + 6 == 10 on every row
nested      : z_nested[t] == sum(sum(x[j] for j in [0:i]) for i in [0:2]);
// nested with the outer range on the index of the constraint :
// [1, 4, 10, 20]
nested_rag  : z_nested_ragged[t] == sum(sum(x[j] for j in [0:i])
                                        for i in [0:t]);
// the body reads the index of the constraint, so it has to be laid out over
// the cells of the sum, unlike every other row above where it appears in the
// range only : t * (1 + .. + x[t]) == [0, 3, 12, 30]
body_uses_t : z_body_t[t] == sum(t * x[i] for i in [0:t]);
#OBJECTIVES
min c : z_ragged[t];
